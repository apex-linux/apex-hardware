# Firmware files that a device needs before it works, by device.
#
# Each line is "<device> <glob>". The glob is relative to /usr/lib/firmware
# and ends in "*" where distributions add a compression suffix. <device> is
# pci:<vendor>:<device> or usb:<vendor>:<product> in lowercase hex, or
# cpu:<vendor_id>:<family> with the family in decimal, as /proc/cpuinfo
# shows it. apex-image checks that the image holds a file for every line;
# the acceptance check looks only at devices that are present.

# AMD Cezanne APU graphics (Radeon Vega, Ryzen 5000 mobile): GFX, SDMA,
# display (DMUB) and video (VCN) microcode. amdgpu does not probe without it.
pci:1002:1638 amdgpu/green_sardine_*

# NVIDIA GA107 (RTX 3050 Mobile): GSP firmware. nouveau cannot drive an
# Ampere GPU without it. The nvidia driver flavor ships its own copy.
pci:10de:25a2 nvidia/ga107/gsp/gsp-*

# Intel Wi-Fi 6 AX200: iwlwifi picks the newest API it supports.
pci:8086:2723 iwlwifi-cc-a0-*.ucode*

# Intel AX200 Bluetooth: operational firmware and DDC parameters.
usb:8087:0029 intel/ibt-20-1-3.sfi*
usb:8087:0029 intel/ibt-20-1-3.ddc*

# AMD family 19h CPUs (Zen 3 and Zen 4): microcode for early loading.
cpu:AuthenticAMD:25 amd-ucode/microcode_amd_fam19h.bin*
